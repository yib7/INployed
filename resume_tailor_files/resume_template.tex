% --- FONT SIZE CONTROL ---
\newcommand{\resumeHeaderSize}{\fontsize{10.5pt}{11.5pt}\selectfont}
\newcommand{\resumeBulletSize}{\fontsize{10pt}{11pt}\selectfont}

% --- BULLET POINTS (all sections) ---
\newcommand{\resumeItem}[1]{
\item\resumeBulletSize{
{#1\vspace{-.75pt}}
}
}

% --- EDUCATION header (two rows: school+location / degree) ---
% #1 = school (bold); #2 = the ", <location>" suffix in plain weight, or empty;
% #3 = dates (right); #4 = the degree line, alone on its own row.
% ATS shape: the school row ends in the same regular font the date column starts
% in, and the degree row has nothing on its right, so a parser that reads a row's
% two cells as one run sees "ST August 2021", never "GPAAugust 2021" or
% "Data Science City". GPA and honors go on their own labelled line
% below (render._education), where "GPA: 3.7/4.0" is the form parsers match.
\newcommand{\resumeSubheading}[4]{
\vspace{-2pt}\item
\begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
\resumeHeaderSize\textbf{#1}#2 & \resumeHeaderSize#3 \\
\resumeHeaderSize\textit{#4} & \\
\end{tabular*}\vspace{-14pt}
}

% --- WORK EXPERIENCE header (one line: title|company left, location|dates right) ---
\newcommand{\resumeSubheadingOneLine}[4]{
\vspace{-2pt}\item
\begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
\resumeHeaderSize\textbf{#1} $|$ \textit{#2} $|$ #3 & \resumeHeaderSize #4 \\
\end{tabular*}\vspace{-7pt}
}

% --- LEADERSHIP header (one row: name left, dates right) ---
\newcommand{\resumeProjectHeading}[2]{
\vspace{-2pt}\item
\begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
\resumeHeaderSize#1 & \resumeHeaderSize#2 \\
\end{tabular*}\vspace{-7pt}
}

% --- PROJECTS header (one line: bold name | link, all left-aligned) ---
% #1 = project name (bolded here); #2 = the " | <link>" suffix, or empty for a
% project with no link. The renderer prints the repo's own host+path as the link
% text (not the word "Link"): an ATS keeps only the extracted text and drops the
% \href target, so a bare "Link" reaches it as a dead word.
\newcommand{\resumeProjectHeadingInline}[2]{
\vspace{-2pt}\item
\begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
\resumeHeaderSize\textbf{#1}#2 & \\
\end{tabular*}\vspace{-7pt}
}
